SQL Query

PostgreSQL expand_less 
SELECT
  product_name,
  SUM(revenue) AS total_revenue,
  (SUM(revenue) - LAG(SUM(revenue)) OVER (PARTITION BY product_id ORDER BY quarter)) / 
  LAG(SUM(revenue)) OVER (PARTITION BY product_id ORDER BY quarter) AS growth_pct
FROM
  sales_data
WHERE
  quarter = 'Q3_2026'
GROUP BY
  product_id, product_name
ORDER BY
  total_revenue DESC
LIMIT 10;
